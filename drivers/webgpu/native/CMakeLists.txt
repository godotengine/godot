cmake_minimum_required(VERSION 3.20)
project(GodotWebGPUDawn LANGUAGES CXX)

# Keep Dawn's own Tint/SPIRV-Tools inside a shared C-API boundary. Godot links
# a separately patched Tint, so linking the two static implementations together
# would violate the one-definition rule even when a linker accepts it.
find_package(Threads REQUIRED)
find_package(Dawn CONFIG REQUIRED)
get_target_property(dawn_includes dawn::dawn_public_config INTERFACE_INCLUDE_DIRECTORIES)
find_path(DAWN_HEADER_DIR dawn/webgpu.h PATHS ${dawn_includes} REQUIRED NO_DEFAULT_PATH)
file(STRINGS "${DAWN_HEADER_DIR}/dawn/webgpu.h" declarations REGEX "^WGPU_EXPORT.* wgpu[A-Za-z0-9_]+\\(")
# Export the same helpers that native_loader.py resolves from dawn_boundary.h.
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/dawn_boundary.h" helpers REGEX "^WGPU[A-Za-z0-9_]* godot_webgpu_[A-Za-z0-9_]+\\(")
set(exports "")
set(elf_exports "{ global:\n")
set(mach_exports "")
foreach(declaration IN LISTS helpers declarations)
    string(REGEX REPLACE ".* ((wgpu|godot_webgpu_)[A-Za-z0-9_]+)\\(.*" "\\1" symbol "${declaration}")
    string(APPEND exports "  ${symbol}\n")
    string(APPEND elf_exports "  ${symbol};\n")
    string(APPEND mach_exports "_${symbol}\n")
endforeach()
string(APPEND elf_exports "local: *; };\n")

add_library(godot_webgpu_dawn SHARED dawn_boundary.cpp)
target_link_libraries(godot_webgpu_dawn PRIVATE dawn::webgpu_dawn ${CMAKE_DL_LIBS})
target_compile_features(godot_webgpu_dawn PRIVATE cxx_std_20)
if(WIN32)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/exports.def" "EXPORTS\n${exports}")
    target_sources(godot_webgpu_dawn PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/exports.def")
elseif(APPLE)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/exports.txt" "${mach_exports}")
    target_link_options(godot_webgpu_dawn PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_BINARY_DIR}/exports.txt")
    set_target_properties(godot_webgpu_dawn PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR ON)
else()
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/exports.map" "${elf_exports}")
    target_link_options(godot_webgpu_dawn PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_BINARY_DIR}/exports.map" "LINKER:-Bsymbolic")
endif()
install(TARGETS godot_webgpu_dawn RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(DIRECTORY "${DAWN_HEADER_DIR}/dawn" "${DAWN_HEADER_DIR}/webgpu" DESTINATION include FILES_MATCHING PATTERN "*.h")
